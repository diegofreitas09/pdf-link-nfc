function ensureMasterMap() {
  if (state.masterMap) return state.masterMap;
  const node = $('master-map');
  if (!node) return null;
  state.masterMap = L.map(node, { zoomControl: true, attributionControl: true, preferCanvas: true })
    .setView(CONFIG.defaultMapCenter, CONFIG.defaultMapZoom);
  L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
    maxZoom: 19,
    updateWhenIdle: false,
    keepBuffer: 4,
    attribution: '&copy; OpenStreetMap contributors'
  }).addTo(state.masterMap);
  const fix = () => state.masterMap?.invalidateSize(true);
  requestAnimationFrame(fix);
  setTimeout(fix, 150);
  setTimeout(fix, 600);
  window.addEventListener('resize', fix, { passive: true });
  return state.masterMap;
}

function masterDivIcon(name, online = false) {
  return L.divIcon({
    className: '',
    html: `<div class="master-marker${online ? ' online' : ' stale'}">${initials(name)}</div>`,
    iconSize: [44,44], iconAnchor: [22,22]
  });
}

function renderMasterMap(focusSelected = false, fitAll = false) {
  if (state.family?.my_role !== 'owner' || $('master-experience')?.classList.contains('hidden')) return;
  const map = ensureMasterMap(); if (!map) return;
  map.invalidateSize(true);
  for (const layer of state.masterLayers) map.removeLayer(layer);
  state.masterLayers = [];
  const points = state.masterDevices.filter((d) => Number.isFinite(d.latitude) && Number.isFinite(d.longitude));
  $('master-map-empty').classList.toggle('hidden', points.length > 0);
  if (!points.length) {
    map.setView(CONFIG.defaultMapCenter, CONFIG.defaultMapZoom);
    setTimeout(() => map.invalidateSize(true), 80);
    return;
  }
  const bounds = [];
  for (const d of points) {
    const latlng = [d.latitude, d.longitude]; bounds.push(latlng);
    const marker = L.marker(latlng, { icon: masterDivIcon(d.display_name || 'M', Boolean(d.is_online)) }).addTo(map);
    marker.bindTooltip(`${d.display_name || 'Membro'} • ${d.is_online ? 'online' : `última posição ${relativeTime(d.captured_at || d.last_seen_at)}`}`, { direction:'top', offset:[0,-20] });
    marker.on('click', () => {
      state.selectedMasterDeviceId = d.device_id;
      renderMasterMembers(); renderMasterDevice();
    });
    state.masterLayers.push(marker);
  }
  const selected = getSelectedMasterDevice();
  if (focusSelected && selected && Number.isFinite(selected.latitude) && Number.isFinite(selected.longitude)) {
    map.flyTo([selected.latitude, selected.longitude], 18, { duration: .55 });
  } else if (fitAll || points.length > 1) {
    map.fitBounds(bounds, { padding:[50,50], maxZoom:16 });
  } else {
    map.setView([points[0].latitude, points[0].longitude], 17);
  }
  setTimeout(() => map.invalidateSize(true), 100);
}

function alertStatusLabel(status) {
  if (status === 'acknowledged') return 'Confirmado';
  if (status === 'dismissed') return 'Dispensado';
  if (status === 'expired') return 'Expirado';
  return 'Aguardando';
}
