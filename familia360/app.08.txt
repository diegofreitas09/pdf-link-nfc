async function loadMasterDevices(showFeedback = false) {
  if (state.family?.my_role !== 'owner') return;
  const { data, error } = await supabase.rpc('get_master_device_dashboard');
  if (error) {
    if (showFeedback) toast(errorMessage(error, 'Não foi possível carregar os aparelhos.'));
    return;
  }
  state.masterDevices = (data || []).filter((item) => item.device_id);
  if (!state.selectedMasterDeviceId || !state.masterDevices.some((d) => d.device_id === state.selectedMasterDeviceId)) {
    state.selectedMasterDeviceId = state.masterDevices[0]?.device_id || null;
  }
  renderMasterMembers();
  renderMasterDevice();
  renderMasterMap();
  await Promise.all([renderMasterAlerts(), loadMasterPlaces(), loadMasterActivity()]);
}

function getSelectedMasterDevice() {
  return state.masterDevices.find((d) => d.device_id === state.selectedMasterDeviceId) || null;
}

function renderMasterMembers() {
  const root = $('master-member-strip');
  if (!root) return;
  root.replaceChildren();
  const onlineCount = state.masterDevices.filter((d) => d.is_online).length;
  $('master-online-summary').textContent = state.masterDevices.length
    ? `${onlineCount} online • ${state.masterDevices.length} aparelho${state.masterDevices.length === 1 ? '' : 's'} vinculado${state.masterDevices.length === 1 ? '' : 's'}`
    : 'Nenhum aparelho vinculado ainda';

  for (const device of state.masterDevices) {
    const btn = document.createElement('button');
    btn.type = 'button';
    btn.className = `master-person-chip${device.device_id === state.selectedMasterDeviceId ? ' active' : ''}`;
    const av = document.createElement('span'); av.className = 'master-chip-avatar'; av.textContent = initials(device.display_name || 'M');
    const copy = document.createElement('span'); copy.className = 'master-chip-copy';
    const name = document.createElement('strong'); name.textContent = device.display_name || 'Membro';
    const status = document.createElement('span'); status.className = device.is_online ? 'online' : ''; status.textContent = device.is_online ? '● Online' : relativeTime(device.last_seen_at);
    copy.append(name, status); btn.append(av, copy);
    btn.addEventListener('click', () => {
      state.selectedMasterDeviceId = device.device_id;
      renderMasterMembers(); renderMasterDevice(); renderMasterMap(true);
    });
    root.append(btn);
  }
}

function permissionLabel(value) {
  if (value === 'granted') return ['Autorizada', 'permission-ok'];
  if (value === 'denied') return ['Negada', 'permission-off'];
  if (value === 'prompt') return ['Aguardando', 'permission-unknown'];
  return ['Não informado', 'permission-unknown'];
}

function setPermission(id, value) {
  const el = $(id); if (!el) return;
  const [label, cls] = permissionLabel(value);
  el.textContent = label;
  el.className = cls;
}
